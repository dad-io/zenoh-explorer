# 2. Launch exactly as the plan row says, leave it idle for 8 s, then kill it (scripts/cargorun.sh)
SPIKE_LOG=$S/t11/cargo-run-idle.log cargo run --offline --example causal_motion_spike
#    -> logs/cargo-run.txt, logs/cargo-run-idle.log (5 frames in 8 s: no repaint while idle)

# 3. Scripted runs (scripts/runall.sh -> scripts/run.sh <profile> <script> <tag> [shots] [rows])
#    The example's autoplay (SPIKE_SCRIPT) calls the same click handler as a real click. It only
#    schedules its own steps with request_repaint_after, so without SPIKE_SHOTS the log shows the
#    effect's real repaint policy. SPIKE_SHOTS reads back every frame with ViewportCommand::Screenshot.
bash run.sh release reveal    rel-reveal                    # timing + policy (logs/rel-reveal.log)
bash run.sh release reveal    rel-reveal-rows2000 "" 2000   # + 2000 label rows in panel B (load proxy)
bash run.sh release supersede rel-supersede                 # clicks at +0, +272, +602 ms (T9 repeat spacing)
bash run.sh release reduced   rel-reduced                   # Reduce motion on before the click
bash run.sh release toggle    rel-toggle                    # Reduce motion on at commit+~400 ms, off at commit+~700 ms
bash run.sh debug   reveal    dev-reveal                    # dev-profile timing
bash run.sh release reveal    shots-reveal 1                # frame-capture runs (GL framebuffer readback)
bash run.sh release supersede shots-supersede 1
bash run.sh release reduced   shots-reduced 1
bash run.sh release toggle    shots-toggle 1
# after rustfmt and the commit, one more timing run on the committed source:
bash run.sh release reveal    rel-reveal-committed

# 4. Analysis. The raw PPM dumps (1760x1040 px per frame) were kept in scratch only and deleted after step 5 (about 5 GB).
python3 analyze.py <logs> > analysis.txt       # frame time, repaint policy, per-side order/onset/peak/settle from the log
python3 extra.py <logs dir> > extra.txt         # interval histograms, shifted cpu_prev, pre-click/at-rest costs, matched-window effect, capture-run cpu
python3 alldiff.py                             # -> diff-{reveal,supersede,reduced,toggle}.txt (diff.py: pixel classes vs the pre-click frame)
python3 sides.py shots-reveal/ logs/shots-reveal.log sides-reveal.txt   # per-side onset and edge colours from pixels
python3 final.py shots-reveal/ shots-supersede/ shots-reduced/ shots-toggle/ > final-frames.txt   # last frames compared
