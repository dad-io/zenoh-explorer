// Logic of zenoh_worker.rs:812-840, condensed but branch-for-branch (exact match checked first at :812)
fn app_match(key_expr: &str, stored_key: &str) -> bool {
    if key_expr == stored_key { return true; }
    if key_expr.contains("**") {
        let prefix = key_expr.trim_end_matches("**").trim_end_matches('/');
        prefix.is_empty() || stored_key.starts_with(prefix)
    } else if key_expr.contains('*') {
        let parts: Vec<&str> = key_expr.split('/').collect();
        let key_parts: Vec<&str> = stored_key.split('/').collect();
        parts.len() == key_parts.len() && parts.iter().zip(key_parts.iter()).all(|(p, kp)| p == &"*" || p == kp)
    } else { false }
}
